SELECT Employee.E_Name
FROM Employee
JOIN Department
ON Employee.E_No = Department.E_no;

SELECT Employee.E_Name, Department.Name
FROM Employee
JOIN Department
ON Employee.E_No = Department.E_no
WHERE Department.Name IN ('HR', 'IT');

SELECT Employee.E_Name
FROM Employee
LEFT JOIN Department
ON Employee.E_No = Department.E_no
WHERE Department.E_no IS NULL;
